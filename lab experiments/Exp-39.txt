#include <stdio.h>
#include <string.h>
#include <ctype.h>

#define MAX 1000

double freq[26] = {
    8.167,1.492,2.782,4.253,12.702,2.228,
    2.015,6.094,6.966,0.153,0.772,4.025,
    2.406,6.749,7.507,1.929,0.095,5.987,
    6.327,9.056,2.758,0.978,2.360,0.150,
    1.974,0.074
};

int main() {
    char cipher[MAX], plain[MAX];
    int top;

    printf("Enter ciphertext: ");
    fgets(cipher, sizeof(cipher), stdin);

    printf("Enter number of top plaintexts: ");
    scanf("%d", &top);

    double scores[26];

    for (int shift = 0; shift < 26; shift++) {
        int count[26] = {0};
        int n = 0;

        for (int i = 0; cipher[i]; i++) {
            if (isalpha((unsigned char)cipher[i])) {
                int x = toupper((unsigned char)cipher[i]) - 'A';
                int p = (x - shift + 26) % 26;

                count[p]++;
                n++;
            }
        }

        double score = 0;

        for (int i = 0; i < 26; i++) {
            double expected = n * freq[i] / 100.0;

            if (expected > 0) {
                double diff = count[i] - expected;
                score += diff * diff / expected;
            }
        }

        scores[shift] = score;
    }

    printf("\nPossible plaintexts:\n");

    for (int rank = 0; rank < top && rank < 26; rank++) {
        int best = -1;
        double min = 1e100;

        for (int i = 0; i < 26; i++) {
            if (scores[i] < min) {
                min = scores[i];
                best = i;
            }
        }

        printf("\nShift %d: ", best);

        for (int i = 0; cipher[i]; i++) {
            if (isalpha((unsigned char)cipher[i])) {
                int x = toupper((unsigned char)cipher[i]) - 'A';
                int p = (x - best + 26) % 26;

                putchar(islower((unsigned char)cipher[i])
                        ? p + 'a' : p + 'A');
            }
            else
                putchar(cipher[i]);
        }

        scores[best] = 1e100;
    }

    return 0;
}
