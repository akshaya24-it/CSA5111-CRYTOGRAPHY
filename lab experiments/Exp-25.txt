#include <stdio.h>

int gcd(int a,int b) {
    while(b) {
        int t=a%b;
        a=b;
        b=t;
    }
    return a;
}

int main() {
    int n,m,g;

    printf("Enter RSA modulus n: ");
    scanf("%d",&n);

    printf("Enter plaintext block m: ");
    scanf("%d",&m);

    g=gcd(n,m);

    if(g>1 && g<n) {
        printf("Factor 1: %d\n",g);
        printf("Factor 2: %d\n",n/g);
    } else {
        printf("No nontrivial factor found.\n");
    }

    return 0;
}
