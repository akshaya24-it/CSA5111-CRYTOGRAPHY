#include <stdio.h>

long long modpow(long long a,long long e,long long n) {
    long long r=1;

    while(e) {
        if(e&1)
            r=r*a%n;

        a=a*a%n;
        e>>=1;
    }

    return r;
}

int main() {
    long long e,n;

    printf("Enter public exponent e: ");
    scanf("%lld",&e);

    printf("Enter modulus n: ");
    scanf("%lld",&n);

    printf("Character Encryption Table:\n");

    for(int m=0;m<26;m++) {
        printf("%c -> %lld\n",
               'A'+m,modpow(m,e,n));
    }

    return 0;
}
