#include <stdio.h>
#include <stdint.h>

void left_shift(unsigned char *in, unsigned char *out, int n) {
    int carry = 0;

    for (int i = n - 1; i >= 0; i--) {
        int next = (in[i] >> 7) & 1;
        out[i] = (in[i] << 1) | carry;
        carry = next;
    }
}

void generate_subkeys(unsigned char *L, unsigned char *K1,
                      unsigned char *K2, int n) {
    unsigned char Rb;

    Rb = (n == 8) ? 0x1B : 0x87;

    left_shift(L, K1, n);

    if (L[0] & 0x80)
        K1[n - 1] ^= Rb;

    left_shift(K1, K2, n);

    if (K1[0] & 0x80)
        K2[n - 1] ^= Rb;
}

int main() {
    unsigned char L[16] = {
        0x12,0x34,0x56,0x78,0x90,0xAB,0xCD,0xEF,
        0x12,0x34,0x56,0x78,0x90,0xAB,0xCD,0xEF
    };

    unsigned char K1[16], K2[16];

    generate_subkeys(L, K1, K2, 16);

    printf("128-bit CMAC Subkeys:\nK1: ");
    for (int i = 0; i < 16; i++)
        printf("%02X", K1[i]);

    printf("\nK2: ");
    for (int i = 0; i < 16; i++)
        printf("%02X", K2[i]);

    printf("\n64-bit Rb constant: 1B");
    printf("\n128-bit Rb constant: 87\n");

    return 0;
}
