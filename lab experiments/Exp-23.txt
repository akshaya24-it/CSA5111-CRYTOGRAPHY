#include <stdio.h>

/* Include the complete S-DES functions:
   keys(), sd(), perm(), fk(), and the S-boxes
   from Program 22. */

int main() {
    int key[10]={0,1,1,1,1,1,1,1,0,1};
    int k1[8],k2[8];

    unsigned char p[]={0x01,0x02,0x04};
    unsigned char c[3],d[3];

    keys(key,k1,k2);

    for(int i=0;i<3;i++)
        c[i]=p[i]^sd(i,k1,k2,0);

    printf("Ciphertext: ");
    for(int i=0;i<3;i++)
        printf("%02X ",c[i]);

    for(int i=0;i<3;i++)
        d[i]=c[i]^sd(i,k1,k2,0);

    printf("\nDecrypted: ");
    for(int i=0;i<3;i++)
        printf("%02X ",d[i]);

    return 0;
}
