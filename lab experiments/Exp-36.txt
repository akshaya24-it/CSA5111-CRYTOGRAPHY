#include <stdio.h>

int gcd(int a, int b) {
    while (b != 0) {
        int t = b;
        b = a % b;
        a = t;
    }
    return a;
}

int main() {
    int a, b;

    char plaintext[100];

    printf("Enter a: ");
    scanf("%d", &a);

    printf("Enter b: ");
    scanf("%d", &b);

    printf("Enter plaintext: ");
    scanf(" %[^\n]", plaintext);

    if (gcd(a, 26) != 1) {
        printf("Invalid key: a must be coprime with 26.\n");
        return 0;
    }

    printf("Ciphertext: ");

    for (int i = 0; plaintext[i] != '\0'; i++) {
        if (plaintext[i] >= 'A' && plaintext[i] <= 'Z') {
            int p = plaintext[i] - 'A';
            int c = (a * p + b) % 26;

            printf("%c", c + 'A');
        }
        else if (plaintext[i] >= 'a' && plaintext[i] <= 'z') {
            int p = plaintext[i] - 'a';
            int c = (a * p + b) % 26;

            printf("%c", c + 'a');
        }
        else {
            printf("%c", plaintext[i]);
        }
    }

    printf("\n");

    return 0;
}
